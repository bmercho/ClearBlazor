@using System.ComponentModel.DataAnnotations
@using System.Globalization
@inherits ClearComponentBase

<CascadingValue Value="this" IsFixed="false">
    <DatePicker BorderThickness=1 CornerRadius=4 @bind-Date=Date1 IsDisabled=true @bind-Date:after=DateChanged />
    <TextBlock>@Date1</TextBlock>

    <DatePicker BorderThickness=1 CornerRadius=4 @bind-Date=Date2 IsReadOnly=true @bind-Date:after=DateChanged />
    <TextBlock>@Date2</TextBlock>
</CascadingValue>

@code {
    DateOnly? Date1 { get; set; } = DateOnly.FromDateTime(DateTime.Now.AddDays(5));
    DateOnly? Date2 { get; set; } = DateOnly.FromDateTime(DateTime.Now.AddDays(-5));

    private async Task DateChanged()
    {
        await Refresh();
    }
}
